\subsection{问题定义与符号说明}
给定一个包含MoE层的视觉语言模型$\mathcal{M}$，专家压缩的目标是在保留任务性能的前提下降低每层专家数量。设压缩保留比例为$r=M/N$，压缩过程可表述为\eqnref{eq:compression_objective}。
\begin{equation}
    \label{eq:compression_objective}
    \mathcal{M}' = \arg\min_{\mathcal{M}'} \; \mathbb{E}_{x \sim \mathcal{D}} \left[ \mathcal{L}(\mathcal{M}', x) \right], \quad \text{s.t.} \; |\mathcal{E}_l'| = M, \; \forall l \in \{1, \dots, L\}
\end{equation}

\subsection{桥接分数计算流程}
桥接分数用于估计专家对图文对应关系的敏感程度。参考DeepSeek-VL2等MoE视觉语言模型的输入组织方式\cite{wu2024deepseekvl2}，测试样例将一次校准样本拆分为四路前向输入：

  \setlength{\itemsep}{\parskip}

  \setlength{\topsep}{0pt}

}
\setlength{\parindent}{\originalparindent}
    \item 原始输入$x$：同时包含匹配的图像和文本信息；
    \item 图像扰动输入$\tilde{x}_{\mathrm{vis}}$：替换图像，文本保持不变；
    \item 文本扰动输入$\tilde{x}_{\mathrm{txt}}$：替换问题文本，图像保持不变；
    \item 图文错配输入$\tilde{x}_{\mathrm{mis}}$：保留问题文本，但替换为另一张图像。
\end{list}

\subsection{算法测试}
\algref{al:admissibility_merge}给出合并场景下的简化伪代码。该算法用于检查algorithm环境在DOCX中是否转换为带标题行、步骤编号和表格边框的样式。

\begin{algorithm}[ht]
    \caption{可容许性约束合并}\label{al:admissibility_merge}
    \begin{algorithmic}[1]
        \State \textbf{输入：}第$l$层专家集合$\mathcal{E}_l$，基础聚类分组$\{\mathcal{G}_1,\dots,\mathcal{G}_K\}$，桥接专家集合$\mathcal{B}_l$
        \State \textbf{输出：}修正后的合并分组$\{\mathcal{G}_1',\dots,\mathcal{G}_K'\}$
        \State 初始化$\mathcal{G}_k' \gets \mathcal{G}_k$，其中$k=1,\dots,K$
        \State 对每个合并组$\mathcal{G}_k'$，计算组内桥接专家集合$\mathcal{B}_k \gets \mathcal{G}_k' \cap \mathcal{B}_l$
        \State 若$\mathcal{B}_k$中少于两个桥接专家，则保留当前分组
        \State 若存在互补冲突，则选择桥接分数较低的专家作为重分配对象
        \State 在其他合并组中寻找不会产生新冲突的目标组
        \State 若目标组存在，则将该专家重分配到目标组
        \State 若目标组不存在，则保留基础聚类结果
        \State 返回$\{\mathcal{G}_1',\dots,\mathcal{G}_K'\}$
    \end{algorithmic}
\end{algorithm}
